import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>

### Run the cycle

On **ACH**, press **Run ACH cycle now** (`runAchCycleNow`).

</Step>
<Step>

### Download

Download the built file from **Files** (`downloadAchFile`).

</Step>
<Step>

### Send

Send it through FedLine ACH outside the app.

</Step>
<Step>

### Record

Record FedACH's file reference and settlement date (`recordAchFileTransmission`).

</Step>
</Steps>
