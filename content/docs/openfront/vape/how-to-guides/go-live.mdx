import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Store details and policies
At `/dashboard/platform/store` enter legal name, a licence line, and the five policies.
</Step>
<Step>
### Merchant profile
At Merchant Profiles complete the draft: PACT registration, reference, digest, origin, dates. Submit and Activate.
</Step>
<Step>
### Jurisdictions and rule sets
Create and activate your launch jurisdictions, then approve and activate one signed rule set per destination. Link jurisdictions to the profile.
</Step>
<Step>
### Licences and provider approvals
Add licences and activate provider approvals for identity, payment, tax, and carrier.
</Step>
<Step>
### Product evidence
Classify variants, add federal authorizations, and publish products.
</Step>
<Step>
### Turn the kill switch off
Use Resume sales on the active profile. Overview should show sales enabled.
</Step></Steps>
