import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Open the fulfill page
From the order press Fulfill.
</Step>
<Step>
### Pick the rate
Choose the rate. Regulated orders list only the approved rate.
</Step>
<Step>
### Buy the label
If it is refused, read the message: it names the missing carrier evidence or approval.
</Step>
<Step>
### Track
Use Track to refresh delivery status.
</Step></Steps>
