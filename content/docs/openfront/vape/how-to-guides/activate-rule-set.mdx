icon: Landmark

import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Create the jurisdiction
Add it with authority sources and activate.
</Step>
<Step>
### Draft the rule set
Use the structured editor, add citations and dates.
</Step>
<Step>
### Approve
A reviewer approves it, recording `approvedAt`.
</Step>
<Step>
### Activate
Activate with the SHA-256 signature digest. Only one active set per destination.
</Step></Steps>
