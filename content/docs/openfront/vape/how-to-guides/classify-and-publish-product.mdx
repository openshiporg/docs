import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Create the product
At Products create a draft with variants and prices.
</Step>
<Step>
### Classify
In Product Evidence press Classify on each variant; set category, nicotine, flavor, and marketing review.
</Step>
<Step>
### Authorize
Add a federal authorization with identifier, evidence digest, and verification date.
</Step>
<Step>
### Publish
Set status to published. Publication fails with the missing evidence if any is absent.
</Step></Steps>
