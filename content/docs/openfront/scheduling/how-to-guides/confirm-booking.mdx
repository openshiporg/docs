description: Work through pending bookings.
icon: CalendarCheck

import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Filter pending
On Bookings open the pending tab.
</Step>
<Step>
### Check payment
Paid events must show payment `paid` before you can confirm.
</Step>
<Step>
### Confirm or decline
Use the row action. Declining cancels the booking.
</Step>
<Step>
### After the appointment
Mark it complete, or no-show if the guest did not attend.
</Step></Steps>
