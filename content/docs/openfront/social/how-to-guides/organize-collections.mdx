import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Create
Press create at `/dashboard/platform/site/collections`. Pick a kind and visibility.
</Step>
<Step>
### Order
Move collections to set the order on your profile.
</Step>
<Step>
### Change visibility
Use `public` to list it and include it in RSS, `unlisted` for link-only, `private` to hide it.
</Step></Steps>
