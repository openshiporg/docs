import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Run creator setup
Open `/dashboard/onboarding`, choose `paper`, `midnight`, or `forest`, and create the starter or a blank site.
</Step>
<Step>
### Edit the profile
At `/dashboard/platform/site` set handle, display name, bio, location, website, image, and logo.
</Step>
<Step>
### Publish
Turn the site visible. Open `/<handle>` to check it.
</Step></Steps>
