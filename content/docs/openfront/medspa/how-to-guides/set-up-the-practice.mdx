import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Sign in
Create the first operator at `/dashboard/init` and run onboarding.
</Step>
<Step>
### Locations and hours
Add a location with timezone in `/platform/locations`.
</Step>
<Step>
### Staff and providers
Invite staff, create provider profiles, set schedules.
</Step>
<Step>
### Catalog
Add categories, treatments and prices, and mark online booking.
</Step>
<Step>
### Publish
Publish [storefront settings](/docs/openfront/medspa/storefront-settings).
</Step>
</Steps>
