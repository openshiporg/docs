title: Publish a listing
description: Get a property onto the public site.
icon: Building2
---

import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>

### Draft
Create the listing at `/dashboard/platform/listings/new`.

</Step>
<Step>

### Media
Add photos by URL under `/images/real-estate/` and set the cover.

</Step>
<Step>

### Authority
Record the license, a signed engagement with marketing authority, and `public_site` publication authority.

</Step>
<Step>

### Activate
Change status to `active` or `coming_soon`. Check `/listings`.

</Step>
</Steps>
