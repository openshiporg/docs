import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>

### Create the deal
On `/dashboard/platform/pipeline` create a deal from the lead and listing.

</Step>
<Step>

### Add the offer
Record price, financing type, contingencies, and response due date.

</Step>
<Step>

### Counter
Create a counter offer. Each links to its parent.

</Step>
<Step>

### Accept
Accept the offer and move the deal to `under_contract`.

</Step>
</Steps>
