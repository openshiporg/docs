import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Create
At `/dashboard/todos/create` enter label and description.
</Step>
<Step>
### Set status
Choose status, priority, tags, and due date.
</Step>
<Step>
### Attach images
Add a cover image and attach images inline or from the library.
</Step>
<Step>
### Assign
With `canManageAllTodos` choose another assignee.
</Step></Steps>
