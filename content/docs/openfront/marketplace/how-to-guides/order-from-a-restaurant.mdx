import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Find a menu
Ask for food. The chat calls `discoverRestaurantMenus`.
</Step>
<Step>
### Add items
Choose modifiers and special instructions.
</Step>
<Step>
### Fulfillment
Pickup needs name, email and phone. Delivery also needs a full address.
</Step>
<Step>
### Pay
Pay in the chat or follow the restaurant checkout link.
</Step>
</Steps>
