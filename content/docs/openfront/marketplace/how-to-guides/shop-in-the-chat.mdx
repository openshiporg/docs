---
title: Shop in the chat
description: Find a product, add it, set shipping and pay.
---

import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
### Ask
Describe what you want and let the chat search stores.
</Step>
<Step>
### Add
Open a result, pick a variant and add to cart.
</Step>
<Step>
### Ship
Enter an address and choose a shipping method.
</Step>
<Step>
### Pay
Pay with Stripe or PayPal, or open the store's checkout link.
</Step>
</Steps>
