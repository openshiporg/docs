import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>

### Place
On `/dashboard/platform/governance` place a hold on a matter, document, or contact.

</Step>
<Step>

### Check
Disposal and anonymization are blocked while the hold is active.

</Step>
<Step>

### Release
Release the hold when it ends.

</Step>
</Steps>
